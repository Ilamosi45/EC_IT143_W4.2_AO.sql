SELECT f.as_of_date AS Monthly_Snapshot, SUM(f.mtd_salary) AS Total_Club_Salary_Spend 
FROM dbo.tblPlayerFact AS f GROUP BY f.as_of_date;


-- PURPOSE: Step 4 - Establish structured database view abstraction layer.
USE EC_IT143_DA;

CREATE VIEW dbo.v_MyFC_salary_summary_AO AS
SELECT f.as_of_date AS Monthly_Snapshot, SUM(f.mtd_salary) AS Total_Club_Salary_Spend 
FROM dbo.tblPlayerFact AS f GROUP BY f.as_of_date;


-- PURPOSE: Step 5.1 - Pipe raw view metadata columns into structural framework storage.
USE EC_IT143_DA;

SELECT Monthly_Snapshot, Total_Club_Salary_Spend INTO dbo.tbl_MyFC_salary_summary_AO FROM dbo.v_MyFC_salary_summary_AO;


-- PURPOSE: Step 5.2 - Refine data limits, typing architecture, and map clustering constraints.
USE EC_IT143_DA;

IF OBJECT_ID('dbo.tbl_MyFC_salary_summary_AO', 'U') IS NOT NULL DROP TABLE dbo.tbl_MyFC_salary_summary_AO;

    Summary_ID INT IDENTITY(1,1) NOT NULL,
    Monthly_Snapshot DATE NOT NULL,
    Total_Club_Salary_Spend DECIMAL(18, 2) NOT NULL,
    CONSTRAINT PK_tbl_MyFC_salary_summary_AO PRIMARY KEY CLUSTERED (Summary_ID ASC)
);
